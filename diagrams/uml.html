    %% PawPal+ core class design.
    %% Owner<->Pet relationship is held by DailyPlan (no Pet.owner back-pointer).

    class Owner {
        +str name
        +int available_minutes
        +str day_start
        +list~str~ preferred_order
    }

    class Pet {
        +str name
        +str species
        +int age_years
    }

    class Task {
        +str title
        +int duration_minutes
        +str priority
        +priority_rank() int
    }

    class ScheduledTask {
        +Task task
        +str start_time
        +str end_time
        +str reason
    }

    class Scheduler {
        +build_plan(owner, pet, tasks) DailyPlan
    }

    class DailyPlan {
        +Owner owner
        +Pet pet
        +list~ScheduledTask~ scheduled
        +list skipped
        +int total_minutes_used
        +summary() str
    }

    Scheduler ..> DailyPlan : creates
    DailyPlan o-- ScheduledTask : contains
    ScheduledTask --> Task : wraps
    DailyPlan --> Owner : for
    DailyPlan --> Pet : for
